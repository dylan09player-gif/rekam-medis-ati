3. Sinkronisasi Google Sheets (Integrasi Real-Time)

Master ICD-10, Stok Obat dan Karyawan tersinkron otomatis 24 Jam di background.

Pengaturan Koneksi

Salin URL Deployment Web App dari Google Apps Script Anda.

Ubah Master ICD-10, Obat, atau Karyawan di Google Sheet lalu klik Sinkronkan agar sistem otomatis terupdate.

Panduan Setup GAS

  1. Buka script.google.com lalu buat project baru
  2. Paste kode di bawah ke Code.gs
  3. Deploy sebagai Web App, Access: Anyone
  4. Salin URL Deployment ke kolom kiri

Struktur Sheet yang dibutuhkan:

  • Sheet "Diagnosis": kolom code, description
  • Sheet "Obat": kolom nama, stok, satuan, kategori
  • Sheet "Karyawan": kolom nikPabrik, nama, dept, gender, tglLahir, hp
// == KLINIK ATI: GOOGLE APPS SCRIPT (TARGETED) ==
var SPREADSHEET_ID = "1sNDmrxb4cB1eYKO-CbBXCWOElOCuiBRdJLG6ERrJkqY";

function getSpreadsheet() {
  try {
    return SpreadsheetApp.openById(SPREADSHEET_ID);
  } catch(e) {
    return SpreadsheetApp.getActiveSpreadsheet();
  }
}

function doGet(e) {
  var ss = getSpreadsheet();
  var result = {};
  var diag = ss.getSheetByName('Diagnosis');
  if (diag) result.icd10 = diag.getDataRange().getValues().slice(1)
    .map(function(r,i){ return {id:'ICD-'+i,code:r[0],description:r[1]}; });
  var obat = ss.getSheetByName('Obat');
  if (obat) result.medicines = obat.getDataRange().getValues().slice(1)
    .map(function(r){ return {nama:r[0],stok:parseInt(r[1])||0,satuan:r[2]||'strip',kategori:r[3]||'Obat'}; });
  var kary = ss.getSheetByName('Karyawan');
  if (kary) result.employees = kary.getDataRange().getValues().slice(1)
    .map(function(r){ return {nikPabrik:r[0],nama:r[1],dept:r[2],gender:r[3],tglLahir:r[4],hp:r[5]}; });
  return ContentService.createTextOutput(JSON.stringify(result))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var data = JSON.parse(e.postData.contents);
  var ss = getSpreadsheet();
  if (data.action === 'pushRecords') {
    var sh = ss.getSheetByName('Rekam Medis') || ss.insertSheet('Rekam Medis');
    if (!sh.getLastRow()) sh.appendRow(['ID','Tgl','NIK','Nama','Dept','Keluhan','Diagnosis','Obat','Nakes']);
    data.records.forEach(function(r){
      sh.appendRow([r.id,r.tanggal,r.nikPabrik,r.namaPasien,r.dept,r.keluhan,r.asesmen,r.plan,r.pemeriksa]);
    });
  }
  return ContentService.createTextOutput(JSON.stringify({success:true}))
    .setMimeType(ContentService.MimeType.JSON);
}

// Fungsi Setup Otomatis Tab & Header di Spreadsheet ini:
function setupTabDanHeader() {
  var ss = getSpreadsheet();
  var d = ss.getSheetByName("Diagnosis") || ss.insertSheet("Diagnosis");
  if (d.getLastRow() === 0) d.appendRow(["code", "description"]);
  
  var o = ss.getSheetByName("Obat") || ss.insertSheet("Obat");
  if (o.getLastRow() === 0) o.appendRow(["nama", "stok", "satuan", "kategori"]);
  
  var k = ss.getSheetByName("Karyawan") || ss.insertSheet("Karyawan");
  if (k.getLastRow() === 0) k.appendRow(["nikPabrik", "nama", "dept", "gender", "tglLahir", "hp"]);
  
  Logger.log("Tab & Header Berhasil Dibuat di RM ATI!");
}
